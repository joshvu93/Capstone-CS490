# Run this batch after applying parameters from GUI

peakstt <- findpeaks(tailtipyfsmcd, minpeakdistance = 20, minpeakheight = 0) # Returns peak Y-axis values and frame id number for major peaks; use 25 for 250 fps and 50 for 500 fps

# TAIL MEAN THRUST POWER

# Enter tail diameter into GUI; td = [enter tail diameter from manifest] (e.g., 0.00205 m) 

# Run this batch after applying parameters from GUI

meantailthrust <- (virtmass*relvel*swimvel*taillatvel)-(0.5*virtmass*(relvel^2)*swimvel) # mean thrust power of tail in Watts[PT = mw UW - ½mw2U]
